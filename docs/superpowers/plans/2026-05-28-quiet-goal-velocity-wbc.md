# Quiet Goal-Velocity WBC Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a navigable, velocity-controlled quiet walking WBC policy for G1, trained from the current velocity policy path with reference-guided quiet gait style and RL quietness proxies, while keeping hooks for later manipulation switching and Bumi transfer.

**Architecture:** Add a new `quiet_goal_velocity_tracking` task beside the existing `velocity_tracking` and `motion_tracking` tasks. The actor only observes deployment-available commands and proprioception: `[move_mask, vx, vy, yaw_rate, quiet_mode]`; reference clips are used for reset, rewards, metrics, and critic-only context, not as actor inputs at inference. Train one PPO policy with task-gated rewards over two task modes: `quiet_style_imitation` and `goal_velocity_generalization`.

**Tech Stack:** HoloMotion, Isaac Lab manager-based envs, Hydra YAML configs, PyTorch/PPO, pytest, conda env `holomotion`. Do not install new packages; if a dependency is missing, stop and report it.

## Design Rules

- Keep existing `velocity_tracking` and `motion_tracking` behavior unchanged.
- Do not train many experts in v1. Train one policy with two sampled task modes.
- Do not use AMP in v1. Reference clips provide kinematic style only; RL rewards decide dynamics and quietness.
- Do not feed raw reference pose or phase to the actor. The actor must remain deployable with joystick velocity and `quiet_mode`.
- Use reference clips generated by Kimodo-like tools only after retargeting and metric filtering.

```python
# holomotion/src/env/isaaclab_components/isaaclab_quiet_goal_command.py
from __future__ import annotations

from dataclasses import MISSING, dataclass
from typing import Sequence

import torch
from isaaclab.utils import configclass

from holomotion.src.env.isaaclab_components.isaaclab_velocity_tracking_command import (
    HoloMotionUniformVelocityCommand,
    HoloMotionUniformVelocityCommandCfg,
    _convert_ranges_dict_to_object,
)


@dataclass
class QuietGoalTaskState:
    num_envs: int
    device: str | torch.device

    def __post_init__(self):
        self.task_name_to_id = {
            "quiet_style_imitation": 0,
            "goal_velocity_generalization": 1,
        }
        self.task_ids = torch.ones(self.num_envs, dtype=torch.long, device=self.device)
        self.quiet_mode = torch.ones(self.num_envs, dtype=torch.float32, device=self.device)

    def assign(self, env_ids: torch.Tensor, imitation_mask: torch.Tensor) -> None:
        env_ids = env_ids.to(device=self.task_ids.device, dtype=torch.long)
        imitation_mask = imitation_mask.to(device=self.task_ids.device, dtype=torch.bool)
        self.task_ids[env_ids] = torch.where(
            imitation_mask,
            torch.zeros_like(env_ids),
            torch.ones_like(env_ids),
        )


class QuietGoalVelocityCommand(HoloMotionUniformVelocityCommand):
    cfg: "QuietGoalVelocityCommandCfg"

    def __init__(self, cfg: "QuietGoalVelocityCommandCfg", env):
        super().__init__(cfg, env)
        self.task_state = QuietGoalTaskState(self.num_envs, self.device)
        env.holo_task_ids = self.task_state.task_ids
        env.holo_task_name_to_id = self.task_state.task_name_to_id

    @property
    def quiet_mode(self) -> torch.Tensor:
        return self.task_state.quiet_mode

    @property
    def task_ids(self) -> torch.Tensor:
        return self.task_state.task_ids

    def _resample_command(self, env_ids: Sequence[int]):
        super()._resample_command(env_ids)
        env_ids_t = torch.as_tensor(env_ids, device=self.device, dtype=torch.long)
        p = float(self.cfg.imitation_prob_initial)
        imitation_mask = torch.rand(env_ids_t.numel(), device=self.device) < p
        self.task_state.assign(env_ids_t, imitation_mask)


@configclass
class QuietGoalVelocityCommandCfg(HoloMotionUniformVelocityCommandCfg):
    class_type: type = QuietGoalVelocityCommand

    quiet_mode_prob: float = 1.0
    imitation_prob_initial: float = 1.0
    imitation_prob_final: float = 0.5
    reference_metadata_path: str = ""
    asset_name: str = MISSING


@configclass
class QuietGoalVelocityCommandsCfg:
    pass


def build_quiet_goal_velocity_commands_config(
    command_config_dict: dict,
) -> QuietGoalVelocityCommandsCfg:
    commands_cfg = QuietGoalVelocityCommandsCfg()
    for name, cfg in command_config_dict.items():
        command_type = cfg.get("type", "QuietGoalVelocityCommandCfg")
        params = cfg.get("params", {}).copy()
        if "ranges" in params and isinstance(params["ranges"], dict):
            params["ranges"] = _convert_ranges_dict_to_object(params["ranges"])
        if command_type != "QuietGoalVelocityCommandCfg":
            raise ValueError(f"Unknown quiet goal velocity command type: {command_type}")
        setattr(commands_cfg, name, QuietGoalVelocityCommandCfg(**params))
    return commands_cfg
```

```python
@staticmethod
def _get_obs_quiet_goal_velocity_command(env: ManagerBasedRLEnv):
    velocity_command = isaaclab_mdp.generated_commands(env, command_name="base_velocity")
    if velocity_command.shape[-1] > 3:
        velocity_command = velocity_command[..., :3]
    move_mask = (velocity_command.norm(dim=-1) > 0.1).to(dtype=velocity_command.dtype)
    command_term = env.command_manager.get_term("base_velocity")
    quiet_mode = command_term.quiet_mode.to(dtype=velocity_command.dtype)
    return torch.cat([move_mask[:, None], velocity_command, quiet_mode[:, None]], dim=-1)


@staticmethod

    command_term = env.command_manager.get_term("base_velocity")
```

```python
def _contact_force_rate_from_history(forces_w_history: torch.Tensor, body_ids: list[int]) -> torch.Tensor:
    forces_z = torch.abs(forces_w_history[:, :, body_ids, 2])
    diff = torch.abs(forces_z[:, 1:] - forces_z[:, :-1])
    return diff.max(dim=1).values.sum(dim=1)


def _touchdown_vertical_speed(
    prev_contact: torch.Tensor,
    cur_contact: torch.Tensor,
    body_vel_z: torch.Tensor,
) -> torch.Tensor:
    new_contact = cur_contact & ~prev_contact
    return torch.sum(torch.clamp(-body_vel_z, min=0.0) * new_contact.to(body_vel_z.dtype), dim=1)
```

```python
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path

import yaml


@dataclass(frozen=True)
class QuietReferenceClip:
    path: str
    robot: str
    mean_vx: float
    mean_vy: float
    mean_yaw_rate: float
    quiet_score: float


@dataclass(frozen=True)
class QuietReferenceIndex:
    clips: tuple[QuietReferenceClip, ...]

    @classmethod
    def from_yaml(cls, path: str | Path) -> "QuietReferenceIndex":
        payload = yaml.safe_load(Path(path).read_text(encoding="utf-8"))
        return cls(tuple(QuietReferenceClip(**item) for item in payload["clips"]))

    def match(
        self,
        vx: float,
        vy: float,
        yaw_rate: float,
        max_lin_error: float,
        max_yaw_error: float,
    ) -> list[QuietReferenceClip]:
        out = []
        for clip in self.clips:
            lin_error = ((clip.mean_vx - vx) ** 2 + (clip.mean_vy - vy) ** 2) ** 0.5
            yaw_error = abs(clip.mean_yaw_rate - yaw_rate)
            if lin_error <= max_lin_error and yaw_error <= max_yaw_error:
                out.append(clip)
        return sorted(out, key=lambda c: c.quiet_score, reverse=True)
```

```python
from __future__ import annotations

import torch


def _mean(value: torch.Tensor) -> float:
    return float(value.detach().float().mean().cpu().item())


def _p95(value: torch.Tensor) -> float:
    return float(torch.quantile(value.detach().float().flatten().cpu(), 0.95).item())


def summarize_quiet_metrics(
    commanded_velocity: torch.Tensor,
    measured_velocity: torch.Tensor,
    touchdown_vz: torch.Tensor,
    peak_normal_force: torch.Tensor,
    force_rate: torch.Tensor,
    foot_slip: torch.Tensor,
    root_jerk: torch.Tensor,
) -> dict[str, float]:
    velocity_error = torch.linalg.norm(commanded_velocity - measured_velocity, dim=-1)
    return {
        "velocity_error_mean": _mean(velocity_error),
        "touchdown_vz_mean": _mean(touchdown_vz),
        "peak_normal_force_p95": _p95(peak_normal_force),
        "force_rate_mean": _mean(force_rate),
        "foot_slip_mean": _mean(foot_slip),
        "root_jerk_mean": _mean(root_jerk),
    }
```

Use Kimodo-like generated motions as kinematic style candidates, not as dynamic truth.

1. Generate low-impact candidate walking clips with short stride, low vertical COM oscillation, low foot touchdown speed, and smooth root motion.
